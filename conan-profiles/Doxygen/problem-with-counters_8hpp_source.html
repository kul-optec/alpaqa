120/// Wraps the given problem into a @ref ProblemWithCounters and keeps track of
121/// how many times each function is called, and how long these calls took.
122/// The wrapper has its own copy of the given problem. Making copies of the
123/// wrapper also copies the underlying problem, but does not copy the evaluation
124/// counters, all copies share the same counters.
125template <class Problem>
126[[nodiscard]] auto problem_with_counters(Problem &&p) {
127 using Prob = std::remove_cvref_t<Problem>;
128 using ProbWithCnt = ProblemWithCounters<Prob>;
129 return ProbWithCnt{std::forward<Problem>(p)};
130}
131
132/// Wraps the given problem into a @ref ProblemWithCounters and keeps track of
133/// how many times each function is called, and how long these calls took.
134/// The wrapper keeps only a reference to the given problem, it is the
135/// responsibility of the caller to make sure that the wrapper does not outlive
136/// the original problem. Making copies of the wrapper does not copy the
137/// evaluation counters, all copies share the same counters.
138template <class Problem>
139[[nodiscard]] auto problem_with_counters_ref(Problem &p) {
140 using Prob = std::remove_cvref_t<Problem>;
141 using ProbWithCnt = ProblemWithCounters<const Prob &>;
142 return ProbWithCnt{p};
143}
